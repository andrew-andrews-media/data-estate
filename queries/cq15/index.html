# CQ15. Which repositories are data products, with their purpose, owner, tier, inputs and consumers?
# One row for each repository typed dprod:DataProduct; its inputs, consumers, projects and issue records
# are each listed in one cell, gathered in their own subqueries so that no gathered value is ever unbound.
# Published at https://w3id.org/data-estate/queries/cq15 under CC BY 4.0.
PREFIX de:      <https://w3id.org/data-estate/vocab#>
PREFIX dcterms: <http://purl.org/dc/terms/>
PREFIX dprod:   <https://ekgf.github.io/dprod/>
PREFIX skos:    <http://www.w3.org/2004/02/skos/core#>

SELECT ?repo ?name ?purpose ?owner ?tier ?target ?inputs ?consumers ?projects ?issueRecords
WHERE {
  ?repo a dprod:DataProduct .
  OPTIONAL { ?repo dcterms:title ?name }
  OPTIONAL { ?repo dprod:purpose ?purpose }
  OPTIONAL { ?repo dprod:dataProductOwner ?o . ?o skos:prefLabel ?owner }
  OPTIONAL { ?repo de:tierNow ?t1 . ?t1 skos:notation ?tier }
  OPTIONAL { ?repo de:tierTarget ?t2 . ?t2 skos:notation ?target }
  OPTIONAL {
    SELECT ?repo (GROUP_CONCAT(DISTINCT ?in; separator=", ") AS ?inputs)
    WHERE { ?repo dprod:inputDataset ?i . ?i dcterms:title ?in }
    GROUP BY ?repo
  }
  OPTIONAL {
    SELECT ?repo (GROUP_CONCAT(DISTINCT ?out; separator=", ") AS ?consumers)
    WHERE { ?c dprod:inputDataset ?repo . ?c dcterms:title ?out }
    GROUP BY ?repo
  }
  OPTIONAL {
    SELECT ?repo (GROUP_CONCAT(DISTINCT ?pt; separator=", ") AS ?projects)
    WHERE { ?pr de:projectRepository ?repo ; dcterms:title ?pt }
    GROUP BY ?repo
  }
  OPTIONAL {
    SELECT ?repo (GROUP_CONCAT(DISTINCT STR(?iss); separator=", ") AS ?issueRecords)
    WHERE { ?repo de:issueRecord ?iss }
    GROUP BY ?repo
  }
}
ORDER BY ?name
