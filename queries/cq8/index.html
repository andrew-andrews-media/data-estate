# CQ8. For each repository, artefact or entity, is what is held declared, inferred or path-only?
# Only artefacts carry a metadata source, so it counts artefacts by theirs, for the whole estate and for each
# repository. Run it over the estate graph merged with the vocabulary, which supplies the concepts and the subclass links.
# Published at https://w3id.org/data-estate/queries/cq8 under CC BY 4.0.
PREFIX de:      <https://w3id.org/data-estate/vocab#>
PREFIX dcterms: <http://purl.org/dc/terms/>
PREFIX rdfs:    <http://www.w3.org/2000/01/rdf-schema#>
PREFIX skos:    <http://www.w3.org/2004/02/skos/core#>

SELECT ?repository ?source (COUNT(DISTINCT ?doc) AS ?n)
WHERE {
  ?doc a/rdfs:subClassOf* de:Artefact ; de:metadataSource ?s .
  OPTIONAL { ?s skos:notation ?sn }
  BIND (COALESCE(STR(?sn), STR(?s)) AS ?source)
  {
    BIND ("(estate)" AS ?repository)
  }
  UNION {
    ?doc dcterms:isPartOf ?repo .
    ?repo dcterms:title ?repository .
  }
}
GROUP BY ?repository ?source
ORDER BY ?repository ?source
