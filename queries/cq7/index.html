# CQ7. How many repositories, artefacts and entities are there, by tenant, domain, tier and class?
# It selects counts and the labels of tenants, domains, tiers and classes, and no other names.
# Run it over the estate graph merged with the vocabulary, which supplies the concepts and the subclass links.
# Published at https://w3id.org/data-estate/queries/cq7 under CC BY 4.0.
PREFIX de:      <https://w3id.org/data-estate/vocab#>
PREFIX dcterms: <http://purl.org/dc/terms/>
PREFIX rdfs:    <http://www.w3.org/2000/01/rdf-schema#>
PREFIX schema:  <https://schema.org/>
PREFIX skos:    <http://www.w3.org/2004/02/skos/core#>

SELECT ?facet ?value (COUNT(DISTINCT ?x) AS ?n)
WHERE {
  {
    { ?x a/rdfs:subClassOf* de:Repository . BIND ("repositories" AS ?value) }
    UNION { ?x a/rdfs:subClassOf* de:Artefact . BIND ("artefacts" AS ?value) }
    UNION {
      VALUES ?c { schema:Organization de:Community schema:Event de:Framework de:Topic schema:Place schema:Person }
      ?x a ?c , skos:Concept .
      BIND ("entities" AS ?value)
    }
    BIND ("0 totals" AS ?facet)
  }
  UNION {
    ?x a/rdfs:subClassOf* de:Repository .
    OPTIONAL { ?x de:tenant/skos:prefLabel ?t }
    BIND (COALESCE(STR(?t), "(none)") AS ?value)
    BIND ("1 repositories by tenant" AS ?facet)
  }
  UNION {
    ?x a/rdfs:subClassOf* de:Repository .
    OPTIONAL { ?x de:domain/skos:prefLabel ?d }
    BIND (COALESCE(STR(?d), "(none)") AS ?value)
    BIND ("2 repositories by domain" AS ?facet)
  }
  UNION {
    ?x a/rdfs:subClassOf* de:Repository .
    OPTIONAL { ?x de:tierNow/skos:notation ?r }
    BIND (COALESCE(STR(?r), "(none)") AS ?value)
    BIND ("3 repositories by tier now" AS ?facet)
  }
  UNION {
    ?x a/rdfs:subClassOf* de:Repository ; de:sensitive ?s .
    BIND (IF(?s, "sensitive", "not sensitive") AS ?value)
    BIND ("4 repositories by sensitive flag" AS ?facet)
  }
  UNION {
    ?x a/rdfs:subClassOf* de:Artefact ; dcterms:isPartOf ?repo .
    OPTIONAL { ?repo de:tenant/skos:prefLabel ?t }
    BIND (COALESCE(STR(?t), "(none)") AS ?value)
    BIND ("5 artefacts by tenant" AS ?facet)
  }
  UNION {
    ?x a/rdfs:subClassOf* de:Artefact .
    OPTIONAL { ?x de:domain/skos:prefLabel ?d }
    BIND (COALESCE(STR(?d), "(none)") AS ?value)
    BIND ("6 artefacts by domain" AS ?facet)
  }
  UNION {
    VALUES ?c { schema:Organization de:Community schema:Event de:Framework de:Topic schema:Place schema:Person }
    ?x a ?c , skos:Concept .
    BIND (REPLACE(STR(?c), "^.*[/#]", "") AS ?value)
    BIND ("7 entities by class" AS ?facet)
  }
}
GROUP BY ?facet ?value
ORDER BY ?facet DESC(?n) ?value
